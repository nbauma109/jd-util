class SwitchPatternMatchingWhen {
    record Point(int x, int y) {}

    static int describe(Object o) {
        return switch (o) {
/*   6 */             case Point(int x, int y) when x > y -> x - y;
/*   7 */             case Point(int x, int y) -> x + y;
/*   8 */             default -> 0;
        };
    }
}
